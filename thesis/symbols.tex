\renewcommand{\nomname}{List of Symbols}
% Symbols listed here appear in at least two distinct \texttt{\textbackslash subsection}
% scopes among Introduction, chapters~2--4, conclusion, and appendix (current body text).

% --- Energy scales ---
\nomenclature[E01]{$E_J$}{Josephson coupling energy of a junction (or effective junction).}
\nomenclature[E02]{$E_C$}{Charging energy, $e^2/(2C_\Sigma)$ for the transmon island.}

% --- Charge and phase ---
\nomenclature[N01]{$n_g$}{Dimensionless offset (gate) charge on the island.}
\nomenclature[P01]{$\varphi$}{Gauge-invariant superconducting phase difference across the junction.}
\nomenclature[P02]{$\Phi_0$}{Superconducting flux quantum, $h/2e$.}
\nomenclature[P03]{$\Phi_{\mathrm{ex}}$}{External magnetic flux threading the circuit (bias and noise coordinate in flux-tunable devices).}

% --- Frequencies and anharmonicity ---
\nomenclature[G01]{$\alpha$}{Leading transmon anharmonicity ($\approx -E_C$); in DRAG, also $\omega_{12}-\omega_{01}$.}
\nomenclature[G02]{$\omega_{01}$}{Transition angular frequency between qubit levels $\ket{0}$ and $\ket{1}$.}
\nomenclature[G03]{$\omega_b$}{Transmon or transmon-like transition angular frequency.}
\nomenclature[G04]{$\omega_d$}{Angular frequency of a parametric flux drive.}
\nomenclature[G05]{$\omega_q$}{Qubit transition angular frequency in the noise model.}

% --- Decoherence ---
\nomenclature[D01]{$T_1$}{Energy relaxation (depolarization) time.}

% --- Noise model ---
\nomenclature[S01]{$B_\lambda$}{First-order system operator coupled to $\delta\lambda$ in $H\approx H_0+B_\lambda\,\delta\lambda$.}
\nomenclature[S02]{$k_B$}{Boltzmann constant.}

\printnomenclature
