%% Spider

\pgfcircdeclarebipolescaled{inductors}
{} % extra anchors
{\ctikzvalof{bipoles/jj/height}} % depth under path line
{spider} % name 
{\ctikzvalof{bipoles/jj/height}} % height above path line
{\ctikzvalof{bipoles/jj/width}} % width
{ % code
   % \pgf@circ@setlinewidth{bipoles}{\pgfstartlinewidth}
   \pgfsetlinewidth{\ctikzvalof{bipoles/jj/thickness}\pgfstartlinewidth}

   % left to right acr
   \pgfpathmoveto{\pgfpoint{\pgf@circ@res@left}{\pgf@circ@res@down}} % start point
   \pgfpathcurveto
    {\pgfpoint{\pgf@circ@res@left}{\pgf@circ@res@up}} % support point 1
    {\pgfpoint{\pgf@circ@res@right}{\pgf@circ@res@down}} % support point 2
    {\pgfpoint{\pgf@circ@res@right}{\pgf@circ@res@up}} % final point
   \pgfusepath{stroke}

   % right to left acr
   \pgfpathmoveto{\pgfpoint{\pgf@circ@res@left}{\pgf@circ@res@up}} % start point
   \pgfpathcurveto
    {\pgfpoint{\pgf@circ@res@left}{\pgf@circ@res@down}} % support point 1
    {\pgfpoint{\pgf@circ@res@right}{\pgf@circ@res@up}} % support point 2
    {\pgfpoint{\pgf@circ@res@right}{\pgf@circ@res@down}} % final point
   \pgfusepath{stroke}

   % straight wire
   \pgfsetlinewidth{\pgfstartlinewidth}
   \pgfpathmoveto{\pgfpoint{\pgf@circ@res@right}{0pt}}
   \pgfpathlineto{\pgfpoint{\pgf@circ@res@left}{0pt}}    
   \pgfusepath{draw}
}

\pgfcirc@activate@bipole@simple{l}{spider}

%% Snail

\pgfcircdeclarebipolescaled{inductors}
{} % extra anchors
{\ctikzvalof{bipoles/jj/height}} % depth under path line
{snail} % name 
{\ctikzvalof{bipoles/jj/height}} % height above path line
{\ctikzvalof{bipoles/jj/width}} % width
{ % code
   % \pgf@circ@setlinewidth{bipoles}{\pgfstartlinewidth}
   \pgfsetlinewidth{\ctikzvalof{bipoles/jj/thickness}\pgfstartlinewidth}

   \pgfplothandlerlineto   
   \pgfplotfunction{\x}{0,10,...,720}{
      \pgfpointxy{
       \ctikzvalof{bipoles/jj/width} *exp(-0.1*rad(\x))*cos(\x)}{
       \ctikzvalof{bipoles/jj/height}*exp(-0.1*rad(\x))*sin(\x)}}
   \pgfusepath{stroke}

   % straight wire
   \pgfsetlinewidth{\pgfstartlinewidth}
   \pgfpathmoveto{\pgfpoint{\pgf@circ@res@right}{0pt}}
   \pgfpathlineto{\pgfpointxy{
      \ctikzvalof{bipoles/jj/width} *exp(-0.1*rad(360))*cos(360)
   }{0pt}}    
   \pgfusepath{draw}
}

\pgfcirc@activate@bipole@simple{l}{snail}

        \pgfusepath{draw}
    \endpgfscope
}
